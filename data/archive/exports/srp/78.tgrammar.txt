((%x)(%y)(%z)(%w)):=(%x)(%y)(%z)(%w);
((%x)(%y)(%z)):=(%x)(%y)(%z);
((%x)(%y)):=(%x)(%y);
;Resolving >BLK (inserting blank spaces)
(%x,>BLK)(%y,^BLK,^PUT,^STAIL):=(%x,->BLK)(" ",+BLK)(%y);
;Resolving >COMMA (inserting comma)
(%x,>COMMA):=(%x,->COMMA)(",",PUT=COMMA,+>BLK);
;Resolving EMARK
(%x,EMARK)(%y,^STAIL):=(%y)(%x);
